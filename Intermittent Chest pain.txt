function escapeHtml(value) {
  return String(value)
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll('"', "&quot;")
    .replaceAll("'", "&#039;");
}

function shuffle(array) {
  const result = array.slice();
  for (let i = result.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [result[i], result[j]] = [result[j], result[i]];
  }
  return result;
}

function renderDecision() {
  const decision = caseData.decisions[state.currentDecision];
  state.selectedOption = null;
  state.reasoning = "";

  const shuffledOptions = shuffle(decision.options);

  app.innerHTML = `
    <section class="screen">
      <h2>${escapeHtml(decision.title)}</h2>

      <div class="card">
        ${decision.update}
        ${decision.vitals ? `
          <div class="vitals">
            ${decision.vitals.map(v => `
              <div class="vital">
                <strong>${escapeHtml(v[0])}</strong>${escapeHtml(v[1])}
              </div>
            `).join("")}
          </div>
        ` : ""}
      </div>

      <div class="options" id="options">
        ${shuffledOptions.map(option => `
          <button class="option" data-option-id="${escapeHtml(option.id)}">
            ${escapeHtml(option.text)}
          </button>
        `).join("")}
      </div>

      <div class="card">
        <label for="reasoning">Why?</label>
        <textarea id="reasoning" placeholder="Enter one or two sentences explaining your reasoning."></textarea>
      </div>

      <button class="primary" id="submitBtn" disabled>Submit choice</button>
    </section>
  `;

  const optionButtons = [...document.querySelectorAll(".option")];
  const reasoningBox = document.getElementById("reasoning");
  const submitBtn = document.getElementById("submitBtn");

  optionButtons.forEach(button => {
    button.addEventListener("click", () => {
      optionButtons.forEach(b => b.classList.remove("selected"));
      button.classList.add("selected");
      state.selectedOption = button.dataset.optionId;
      updateSubmitState();
    });
  });

  reasoningBox.addEventListener("input", () => {
    state.reasoning = reasoningBox.value;
    updateSubmitState();
  });

  function updateSubmitState() {
    submitBtn.disabled =
      !state.selectedOption ||
      !state.reasoning.trim();
  }

  submitBtn.addEventListener("click", () => {
    const option = decision.options.find(o => o.id === state.selectedOption);
    const reasoning = state.reasoning.trim();

    state.path.push({
      decision: decision.title,
      option: option.text,
      classification: option.classification,
      reasoning
    });

    state.pendingConsequence = option;
    renderConsequence();
  });
}

function renderConsequence() {
  const option = state.pendingConsequence;

  const labelClass = option.classification.toLowerCase();

  app.innerHTML = `
    <section class="screen">
      <span class="consequence-label ${escapeHtml(labelClass)}">
        ${escapeHtml(option.classification)}
      </span>

      <div class="card">
        <p>${escapeHtml(option.outcome)}</p>
        ${option.reason ? `<p>${escapeHtml(option.reason)}</p>` : ""}
      </div>

      <div class="next-decision">
        Next decision: ${escapeHtml(option.next)}
      </div>

      <button class="primary" id="continueBtn">Continue</button>
    </section>
  `;

  document.getElementById("continueBtn").addEventListener("click", () => {
    if (state.currentDecision < caseData.decisions.length - 1) {
      state.currentDecision += 1;
      renderDecision();
    } else {
      renderDebrief();
    }
  });
}

      <div class="button-row">
        <button class="primary" id="downloadBtn">Download results for your instructor</button>
        <button class="secondary" id="restartBtn">Start over</button>
      </div>
    </section>
  `;

  document.getElementById("downloadBtn").addEventListener("click", downloadResults);
  document.getElementById("restartBtn").addEventListener("click", () => {
    state = {
      currentDecision: 0,
      selectedOption: null,
      reasoning: "",
      path: [],
      pendingConsequence: null
    };
    renderStart();
  });
}

function downloadResults() {
  const now = new Date();
  const dateString =
    now.getFullYear() + "-" +
    String(now.getMonth() + 1).padStart(2, "0") + "-" +
    String(now.getDate()).padStart(2, "0");

  const counts = state.path.reduce((acc, entry) => {
    acc[entry.classification.toLowerCase()]++;
    return acc;
  }, { optimal: 0, acceptable: 0, problematic: 0 });

  let text = "";
  text += "Case: " + CASE_TITLE + "\n";
  text += "Date: " + dateString + "\n";
  text += "\n";

  state.path.forEach((entry, index) => {
    text += "Step " + (index + 1) + "\n";
    text += "Decision: " + entry.decision + "\n";
    text += "Option chosen: " + entry.option + "\n";
    text += "Classification: " + entry.classification + "\n";
    text += "Learner reasoning: " + entry.reasoning + "\n";
    text += "\n";
  });

  text += "Summary: " +
    "Optimal: " + counts.optimal +
    ", Acceptable: " + counts.acceptable +
    ", Problematic: " + counts.problematic + "\n";

  const blob = new Blob([text], { type: "text/plain;charset=utf-8" });
  const url = URL.createObjectURL(blob);
  const anchor = document.createElement("a");

  anchor.href = url;
  anchor.download = "case-results.txt";
  document.body.appendChild(anchor);
  anchor.click();
  anchor.remove();

  setTimeout(() => URL.revokeObjectURL(url), 1000);
}

renderStart();
